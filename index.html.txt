        if (navigator.share) {
          try { await navigator.share(data); } catch (e) { /* cancelado */ }
        } else if (navigator.clipboard) {
          try {
            await navigator.clipboard.writeText(PAGE_URL);
            toast('Link copiado');
          } catch (e) { toast('Não foi possível copiar o link'); }
        }
      });

      /* ---------- Áreas de atuação: cresce, balança e para ---------- */
      const detailEl = document.getElementById('areaDetail');
      const tags = document.querySelectorAll('.tag');

      function wiggle(tag) {
        if (!window.anime || reduceMotion) return;
        anime.remove(tag);
        anime({
          targets: tag,
          scale: [{ value: 1.16, duration: 260, easing: 'easeOutBack' }, { value: 1.1, duration: 300 }],
          rotate: [{ value: -7, duration: 110 }, { value: 6, duration: 110 }, { value: -3, duration: 110 }, { value: 0, duration: 160 }],
          translateY: [{ value: -4, duration: 260 }, { value: -2, duration: 300 }],
          easing: 'easeOutQuad'
        });
      }

      function settle(tag) {
        if (!window.anime || reduceMotion) return;
        anime.remove(tag);
        anime({ targets: tag, scale: 1, rotate: 0, translateY: 0, duration: 450, easing: 'easeOutElastic(1, .6)' });
      }

      function showDetail(text) {
        if (!window.anime || reduceMotion) { detailEl.textContent = text; return; }
        anime({
          targets: detailEl, opacity: 0, translateY: 6, duration: 150, easing: 'easeInQuad',
          complete: () => {
            detailEl.textContent = text;
            anime({ targets: detailEl, opacity: 1, translateY: 0, duration: 350, easing: 'easeOutQuad' });
          }
        });
      }

      tags.forEach(tag => {
        tag.addEventListener('mouseenter', () => wiggle(tag));
        tag.addEventListener('mouseleave', () => { if (!tag.classList.contains('is-active')) settle(tag); });
        tag.addEventListener('click', () => {
          tags.forEach(t => {
            if (t !== tag) { t.classList.remove('is-active'); t.setAttribute('aria-pressed', 'false'); settle(t); }
          });
          tag.classList.add('is-active');
          tag.setAttribute('aria-pressed', 'true');
          wiggle(tag);
          showDetail(tag.dataset.detail);
          scaleApi && scaleApi.nudge(tag);
        });
      });

      /* ---------- Balança da justiça em 3D (Three.js) ---------- */
      let scaleApi = null;
      if (window.THREE) scaleApi = buildScale();

      function buildScale() {
        const stage = document.getElementById('stage');
        const canvas = document.getElementById('scale3d');
        let renderer;
        try {
          renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true });
        } catch (e) { return null; }
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        renderer.outputEncoding = THREE.sRGBEncoding;

        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(35, 1, 0.1, 100);
        camera.position.set(0, 0, 14);

        scene.add(new THREE.HemisphereLight(0xfff6e8, 0x1a1b1e, 0.9));
        const key = new THREE.DirectionalLight(0xffffff, 1.3);
        key.position.set(4, 6, 8);
        scene.add(key);
        const rim = new THREE.PointLight(0xe8d9bd, 1.2, 30);
        rim.position.set(-6, 2, 4);
        scene.add(rim);

        const gold = new THREE.MeshStandardMaterial({ color: 0xC8BBA3, metalness: 0.65, roughness: 0.28 });
        const dark = new THREE.MeshStandardMaterial({ color: 0x55524c, metalness: 0.7, roughness: 0.35 });
        const lineMat = new THREE.LineBasicMaterial({ color: 0xC8BBA3, transparent: true, opacity: 0.7 });

        const root = new THREE.Group();
        scene.add(root);

        // Coluna central (aparece acima e abaixo da foto)
        const pillar = new THREE.Mesh(new THREE.CylinderGeometry(0.07, 0.1, 7.2, 24), gold);
        pillar.position.y = -0.55;
        root.add(pillar);

        const finial = new THREE.Mesh(new THREE.SphereGeometry(0.2, 32, 16), gold);
        finial.position.y = 3.45;
        root.add(finial);

        const base = new THREE.Mesh(new THREE.CylinderGeometry(0.55, 0.85, 0.22, 48), dark);
        base.position.y = -4.15;
        root.add(base);
        const base2 = new THREE.Mesh(new THREE.CylinderGeometry(0.3, 0.5, 0.2, 48), gold);
        base2.position.y = -3.95;
        root.add(base2);

        // Travessão que inclina
        const beam = new THREE.Group();
        beam.position.y = 2.95;
        root.add(beam);

        const bar = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.05, 7.6, 16), gold);
        bar.rotation.z = Math.PI / 2;
        beam.add(bar);

        const hub = new THREE.Mesh(new THREE.TorusGeometry(0.22, 0.05, 16, 48), gold);
        beam.add(hub);

        const PAN_DROP = 3.3;
        const pans = [-1, 1].map(side => {
          const tip = new THREE.Mesh(new THREE.SphereGeometry(0.1, 16, 12), gold);
          tip.position.x = side * 3.8;
          beam.add(tip);

          const pivot = new THREE.Group();
          pivot.position.x = side * 3.8;
          beam.add(pivot);

          const pan = new THREE.Mesh(
            new THREE.SphereGeometry(0.95, 48, 16, 0, Math.PI * 2, Math.PI * 0.62, Math.PI * 0.38),
            gold
          );
          pan.material.side = THREE.DoubleSide;
          pan.position.y = -PAN_DROP + 0.72;
          pivot.add(pan);

          // Correntes (3 fios até a borda do prato)
          const rimY = -PAN_DROP + 0.72 + Math.cos(Math.PI * 0.62) * 0.95;
          const rimR = Math.sin(Math.PI * 0.62) * 0.95;
          for (let i = 0; i < 3; i++) {
            const a = (i / 3) * Math.PI * 2 + Math.PI / 6;
            const geo = new THREE.BufferGeometry().setFromPoints([
              new THREE.Vector3(0, 0, 0),
              new THREE.Vector3(Math.cos(a) * rimR, rimY, Math.sin(a) * rimR)
            ]);
            pivot.add(new THREE.Line(geo, lineMat));
          }
          return { pivot, side, sway: 0, swayVel: 0 };
        });

        // Tamanho responsivo
        function resize() {
          const w = stage.clientWidth, h = stage.clientHeight;
          renderer.setSize(w, h, false);
          camera.aspect = w / h;
          // Afasta a câmera em telas estreitas para os pratos caberem
          camera.position.z = w < 380 ? 14 * (380 / w) : 14;
          camera.updateProjectionMatrix();
        }
        resize();
        if ('ResizeObserver' in window) new ResizeObserver(resize).observe(stage);
        else window.addEventListener('resize', resize);

        // Entrada: mouse ou giroscópio
        const target = { x: 0, y: 0 };
        const current = { x: 0, y: 0 };

        window.addEventListener('pointermove', e => {
          if (e.pointerType === 'touch') return;
          target.x = (e.clientX / window.innerWidth) * 2 - 1;
          target.y = (e.clientY / window.innerHeight) * 2 - 1;
        });

        let baseBeta = null;
        function onOrient(e) {
          if (e.gamma == null) return;
          if (baseBeta === null) baseBeta = e.beta;
          target.x = Math.max(-1, Math.min(1, e.gamma / 30));
          target.y = Math.max(-1, Math.min(1, (e.beta - baseBeta) / 30));
        }

        function enableGyro() {
          if (typeof DeviceOrientationEvent === 'undefined') return;
          if (typeof DeviceOrientationEvent.requestPermission === 'function') {
            // iPhone: precisa de permissão após um toque
            DeviceOrientationEvent.requestPermission()
              .then(s => { if (s === 'granted') window.addEventListener('deviceorientation', onOrient); })
              .catch(() => {});
          } else {
            window.addEventListener('deviceorientation', onOrient);
          }
        }
        if (typeof DeviceOrientationEvent !== 'undefined' &&
            typeof DeviceOrientationEvent.requestPermission !== 'function') {
          enableGyro();
        } else {
          window.addEventListener('touchend', enableGyro, { once: true });
        }

        // Estado da inclinação do travessão (mola)
        let tilt = 0, tiltVel = 0, impulse = 0;

        const intro = { s: reduceMotion ? 1 : 0.001 };
        root.scale.setScalar(intro.s);
        if (window.anime && !reduceMotion) {
          anime({ targets: intro, s: 1, duration: 1600, delay: 250, easing: 'easeOutElastic(1, .7)' });
        }

        const clock = new THREE.Clock();
        let visible = true;
        document.addEventListener('visibilitychange', () => { visible = !document.hidden; });

        function frame() {
          requestAnimationFrame(frame);
          if (!visible) return;
          const t = clock.getElapsedTime();

          current.x += (target.x - current.x) * 0.06;
          current.y += (target.y - current.y) * 0.06;

          const idle = reduceMotion ? 0 : 1;
          root.scale.setScalar(intro.s);
          root.rotation.y = current.x * 0.6 + Math.sin(t * 0.4) * 0.08 * idle;
          root.rotation.x = current.y * 0.22;

          // A balança "pesa" para o lado do mouse / inclinação
          const goal = -current.x * 0.14 + Math.sin(t * 0.7) * 0.02 * idle + impulse;
          tiltVel += (goal - tilt) * 0.08;
          tiltVel *= 0.86;
          tilt += tiltVel;
          impulse *= 0.9;
          beam.rotation.z = tilt;

          // Pratos ficam na vertical e balançam com inércia
          pans.forEach(p => {
            p.swayVel += (-tiltVel * 1.6 - p.sway * 0.05);
            p.swayVel *= 0.92;
            p.sway += p.swayVel * 0.5;
            p.pivot.rotation.z = -tilt + p.sway;
            p.pivot.rotation.x = Math.sin(t * 0.9 + p.side) * 0.03 * idle - current.y * 0.1;
          });

          renderer.render(scene, camera);
        }
        frame();

        return {
          // Tocar numa área dá um "empurrão" na balança
          nudge(el) {
            const r = el.getBoundingClientRect();
            const cx = (r.left + r.width / 2) / window.innerWidth;
            impulse = cx < 0.5 ? 0.22 : -0.22;
          }
        };
      }

      /* ---------- Animação de entrada (Anime.js) ---------- */
      const all = document.querySelectorAll('[data-reveal]');
      const ring = document.querySelector('.ring circle');

      if (!window.anime || reduceMotion) {
        all.forEach(el => el.style.opacity = 1);
        return;
      }

      const ringLength = 2 * Math.PI * 85;
      ring.style.strokeDasharray = ringLength;
      ring.style.strokeDashoffset = ringLength;

      const tl = anime.timeline({ easing: 'cubicBezier(.22, 1, .36, 1)' });

      tl.add({
        targets: '[data-reveal="photo"]',
        opacity: [0, 1],
        scale: [0.92, 1],
        duration: 900
      })
      .add({
        targets: ring,
        strokeDashoffset: [ringLength, 0],
        duration: 1400,
        easing: 'easeInOutQuart'
      }, 150)
      .add({
        targets: '.identity > [data-reveal]:not([data-reveal="photo"])',
        opacity: [0, 1],
        translateY: [14, 0],
        duration: 800,
        delay: anime.stagger(110)
      }, 450)
      .add({
        targets: '[data-reveal="tag"]',
        opacity: [0, 1],
        scale: [0.85, 1],
        duration: 600,
        delay: anime.stagger(80)
      }, 750)
      .add({
        targets: '[data-reveal="btn"]',
        opacity: [0, 1],
        translateY: [18, 0],
        duration: 800,
        delay: anime.stagger(90)
      }, 950)
      .add({
        targets: '[data-reveal="qr"], footer[data-reveal]',
        opacity: [0, 1],
        translateY: [18, 0],
        duration: 900,
        delay: anime.stagger(120)
      }, 1350);

      /* Micro-interação de toque (mobile) */
      document.querySelectorAll('.btn').forEach(btn => {
        btn.addEventListener('touchstart', () => {
          anime({ targets: btn.querySelector('.icon, .ico'), scale: [1, 1.12, 1], duration: 450, easing: 'easeOutQuad' });
        }, { passive: true });
      });
    })();
  </script>
